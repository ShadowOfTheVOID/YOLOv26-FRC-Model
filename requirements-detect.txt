# the scouting API import nothing but the standard library and must keep
# working on a host that has none of this.
#
#   python3 -m venv .venv-detect
#   .venv-detect/bin/pip install -r requirements-detect.txt
#   .venv-detect/bin/python run.py detect --weights runs/<name>/weights/best.pt
#
# On a CPU-only box install torch from the CPU index first, or pip pulls the
# CUDA build and several gigabytes with it:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
